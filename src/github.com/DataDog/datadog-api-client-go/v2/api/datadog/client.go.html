    61  // UseDelegatedTokenAuth sets the Authorization header with a delegated token if available in the context.
    62  func UseDelegatedTokenAuth(ctx context.Context, headerParams *map[string]string, delegatedTokenConfig *DelegatedTokenConfig) error {
    63  	if ctx != nil {
    64  		if delegatedTokenCreds, ok := ctx.Value(ContextDelegatedToken).(*DelegatedTokenCredentials); ok {
    65  			if delegatedTokenCreds.DelegatedToken == "" || time.Now().After(delegatedTokenCreds.Expiration) {
    66  				newCreds, err := CallDelegatedTokenAuthenticate(ctx, delegatedTokenConfig)
    67  				if err != nil {
    68  					log.Printf("Failed to retrieve delegated token: %v", err)
    69  					// Reset the token if authentication failed
    70  					delegatedTokenCreds.DelegatedToken = ""
    71  					return err
    72  				}
    73  				delegatedTokenCreds.DelegatedToken = newCreds.DelegatedToken
    74  				delegatedTokenCreds.DelegatedProof = newCreds.DelegatedProof
    75  				delegatedTokenCreds.OrgUUID = newCreds.OrgUUID
    76  				delegatedTokenCreds.Expiration = newCreds.Expiration
    77  			}
    78  			// If authentication succeeded use delegated token auth
    79  			if delegatedTokenCreds.DelegatedToken != "" {
    80  				(*headerParams)[authorizationHeader] = fmt.Sprintf(bearerTokenFormat, delegatedTokenCreds.DelegatedToken)
    81  			}
    82  		} else {
    83  			return errors.New("DelegatedTokenCredentials not found in context")
    84  		}
    85  	}
    86  	return nil
    87  }

   128  // ParameterToString convert interface{} parameters to string, using a delimiter if format is provided.
   129  func ParameterToString(obj interface{}, collectionFormat string) string {
   130  	var delimiter string
   131  
   132  	switch collectionFormat {
   133  	case "pipes":
   134  		delimiter = "|"
   135  	case "ssv":
   136  		delimiter = " "
   137  	case "tsv":
   138  		delimiter = "\t"
   139  	case "csv":
   140  		delimiter = ","
   141  	}
   142  
   143  	if reflect.TypeOf(obj).Kind() == reflect.Slice {
   144  		return strings.Trim(strings.ReplaceAll(fmt.Sprint(obj), " ", delimiter), "[]")
   145  	} else if t, ok := obj.(time.Time); ok {
   146  		if t.Nanosecond() == 0 {
   147  			return t.Format("2006-01-02T15:04:05Z07:00")
   148  		}
   149  		return t.Format("2006-01-02T15:04:05.000Z07:00")
   150  	}
   151  
   152  	return fmt.Sprintf("%v", obj)
   153  }

   160  // CallAPI do the request.
   161  func (c *APIClient) CallAPI(request *http.Request) (*http.Response, error) {
   162  	var rawBody []byte
   163  	if request.Body != nil && request.Body != http.NoBody {
   164  		rawBody, _ = io.ReadAll(request.Body)
   165  		request.Body.Close()
   166  	}
   167  	ctx, ccancel := context.WithTimeout(request.Context(), c.Cfg.RetryConfiguration.HTTPRetryTimeout)
   168  	defer ccancel()
   169  	retryCount := 0
   170  	for {
   171  		newRequest := copyRequest(request, &rawBody)
   172  		if c.Cfg.Debug {
   173  			dump, err := httputil.DumpRequestOut(newRequest, true)
   174  			if err != nil {
   175  				return nil, err
   176  			}
   177  			// Strip any credential values from the request being logged.
   178  			// ContextAPIKeys carries the DD-API-KEY / DD-APPLICATION-KEY values;
   179  			// ContextAccessToken carries bearer tokens (PATs, delegated tokens).
   180  			var secretsToRedact []string
   181  			if keys, ok := newRequest.Context().Value(ContextAPIKeys).(map[string]APIKey); ok && keys != nil {
   182  				for _, apiKey := range keys {
   183  					if apiKey.Key != "" {
   184  						secretsToRedact = append(secretsToRedact, apiKey.Key)
   185  					}
   186  				}
   187  			}
   188  			if token, ok := newRequest.Context().Value(ContextAccessToken).(string); ok && token != "" {
   189  				secretsToRedact = append(secretsToRedact, token)
   190  			}
   191  			for _, secret := range secretsToRedact {
   192  				valueRegex := regexp.MustCompile(fmt.Sprintf("(?m)%s", regexp.QuoteMeta(secret)))
   193  				dump = valueRegex.ReplaceAll(dump, []byte("REDACTED"))
   194  			}
   195  			log.Printf("\n%s\n", string(dump))
   196  		}
   197  		resp, requestErr := c.Cfg.HTTPClient.Do(newRequest)
   198  
   199  		if requestErr != nil {
   200  			return resp, requestErr
   201  		}
   202  
   203  		if c.Cfg.Debug {
   204  			dump, _ := httputil.DumpResponse(resp, true)
   205  			if c.Cfg.RetryConfiguration.EnableRetry {
   206  				log.Println("Max retries:", c.Cfg.RetryConfiguration.MaxRetries, " Current retry:", retryCount)
   207  				if retryCount == c.Cfg.RetryConfiguration.MaxRetries {
   208  					log.Println("Max retries reached")
   209  				}
   210  			}
   211  			log.Printf("\n%s\n", string(dump))
   212  		}
   213  
   214  		retryDuration, shouldRetry := c.shouldRetryRequest(resp, retryCount)
   215  		if !shouldRetry {
   216  			return resp, requestErr
   217  		}
   218  		if deadline, ok := ctx.Deadline(); ok && *retryDuration >= time.Until(deadline) {
   219  			return resp, requestErr
   220  		}
   221  
   222  		if !retryDelayElapsed(ctx, time.After(*retryDuration)) {
   223  			return resp, requestErr
   224  		}
   225  		if resp.Body != nil {
   226  			resp.Body.Close()
   227  		}
   228  		retryCount++
   229  		continue
   230  
   231  	}
   232  }

   236  	case <-ctx.Done():
   237  		return false
   238  	case <-delay:
   239  		return ctx.Err() == nil

   243  // Determine if a request should be retried
   244  func (c *APIClient) shouldRetryRequest(response *http.Response, retryCount int) (*time.Duration, bool) {
   245  	enableRetry := c.Cfg.RetryConfiguration.EnableRetry
   246  	maxRetries := c.Cfg.RetryConfiguration.MaxRetries
   247  	if !enableRetry || retryCount >= maxRetries {
   248  		return nil, false
   249  	}
   250  	isRateLimited := response.StatusCode == http.StatusTooManyRequests
   251  	isServerError := response.StatusCode >= http.StatusInternalServerError
   252  	isRetryableStatusCode := isRateLimited || isServerError
   253  	if !isRetryableStatusCode {
   254  		return nil, false
   255  	}
   256  
   257  	retryVal := c.Cfg.RetryConfiguration.BackOffBase * math.Pow(c.Cfg.RetryConfiguration.BackOffMultiplier, float64(retryCount))
   258  	retryDuration := time.Duration(retryVal) * time.Second
   259  
   260  	if isRateLimited {
   261  		retryDuration = max(retryDuration, parseRateLimitReset(response.Header.Get(rateLimitResetHeader)))
   262  	}
   263  
   264  	retryDuration = addRetryJitter(retryDuration, c.retryJitter())
   265  	return &retryDuration, true
   266  }

   268  func parseRateLimitReset(value string) time.Duration {
   269  	const maxResetSeconds = math.MaxInt64 / int64(time.Second)
   270  
   271  	seconds, err := strconv.ParseInt(value, 10, 64)
   272  	if err != nil || seconds < 0 || seconds > maxResetSeconds {
   273  		return 0
   274  	}
   275  	return time.Duration(seconds) * time.Second
   276  }

   278  func addRetryJitter(retryDuration, jitter time.Duration) time.Duration {
   279  	if jitter > time.Duration(math.MaxInt64)-retryDuration {
   280  		return time.Duration(math.MaxInt64)
   281  	}
   282  	return retryDuration + jitter
   283  }

   299  // PrepareRequest build the request.
   300  func (c *APIClient) PrepareRequest(
   301  	ctx context.Context,
   302  	path string, method string,
   303  	postBody interface{},
   304  	headerParams map[string]string,
   305  	queryParams url.Values,
   306  	formParams url.Values,
   307  	formFile *FormFile) (localVarRequest *http.Request, err error) {
   308  
   309  	var body *bytes.Buffer
   310  
   311  	// Apply default headers unless they are overridden
   312  	for header, value := range c.Cfg.DefaultHeader {
   313  		if _, exists := headerParams[header]; !exists {
   314  			headerParams[header] = value
   315  		}
   316  	}
   317  
   318  	// Detect postBody type and post.
   319  	if postBody != nil {
   320  		contentType := headerParams["Content-Type"]
   321  		if contentType == "" {
   322  			contentType = detectContentType(postBody)
   323  			headerParams["Content-Type"] = contentType
   324  		}
   325  
   326  		body, err = setBody(postBody, contentType)
   327  		if err != nil {
   328  			return nil, err
   329  		}
   330  	}
   331  
   332  	// add form parameters and file if available.
   333  	if strings.HasPrefix(headerParams["Content-Type"], "multipart/form-data") && len(formParams) > 0 || formFile != nil {
   334  		if body != nil {
   335  			return nil, errors.New("cannot specify postBody and multipart form at the same time")
   336  		}
   337  		body = &bytes.Buffer{}
   338  		w := multipart.NewWriter(body)
   339  
   340  		for k, v := range formParams {
   341  			for _, iv := range v {
   342  				if strings.HasPrefix(k, "@") { // file
   343  					err = addFile(w, k[1:], iv)
   344  					if err != nil {
   345  						return nil, err
   346  					}
   347  				} else { // form value
   348  					_ = w.WriteField(k, iv)
   349  				}
   350  			}
   351  		}
   352  		if formFile != nil {
   353  			w.Boundary()
   354  			part, err := w.CreateFormFile(formFile.FormFileName, filepath.Base(formFile.FileName))
   355  			if err != nil {
   356  				return nil, err
   357  			}
   358  			_, err = part.Write(formFile.FileBytes)
   359  			if err != nil {
   360  				return nil, err
   361  			}
   362  		}
   363  
   364  		// Set the Boundary in the Content-Type
   365  		headerParams["Content-Type"] = w.FormDataContentType()
   366  
   367  		// Set Content-Length
   368  		headerParams["Content-Length"] = fmt.Sprintf("%d", body.Len())
   369  		w.Close()
   370  	}
   371  
   372  	if strings.HasPrefix(headerParams["Content-Type"], "application/x-www-form-urlencoded") && len(formParams) > 0 {
   373  		if body != nil {
   374  			return nil, errors.New("cannot specify postBody and x-www-form-urlencoded form at the same time")
   375  		}
   376  		body = &bytes.Buffer{}
   377  		body.WriteString(formParams.Encode())
   378  		// Set Content-Length
   379  		headerParams["Content-Length"] = fmt.Sprintf("%d", body.Len())
   380  	}
   381  
   382  	// Setup path and query parameters
   383  	url, err := url.Parse(path)
   384  	if err != nil {
   385  		return nil, err
   386  	}
   387  
   388  	// Override request host, if applicable
   389  	if c.Cfg.Host != "" {
   390  		url.Host = c.Cfg.Host
   391  	}
   392  
   393  	// Override request scheme, if applicable
   394  	if c.Cfg.Scheme != "" {
   395  		url.Scheme = c.Cfg.Scheme
   396  	}
   397  
   398  	// Adding Query Param
   399  	query := url.Query()
   400  	for k, v := range queryParams {
   401  		for _, iv := range v {
   402  			query.Add(k, iv)
   403  		}
   404  	}
   405  
   406  	// Encode the parameters.
   407  	url.RawQuery = query.Encode()
   408  
   409  	// Generate a new request
   410  	if body != nil {
   411  		if headerParams["Content-Encoding"] == "gzip" {
   412  			var buf bytes.Buffer
   413  			compressor := gzip.NewWriter(&buf)
   414  			if _, err = compressor.Write(body.Bytes()); err != nil {
   415  				return nil, err
   416  			}
   417  			if err = compressor.Close(); err != nil {
   418  				return nil, err
   419  			}
   420  			body = &buf
   421  
   422  		} else if headerParams["Content-Encoding"] == "deflate" {
   423  			var buf bytes.Buffer
   424  			compressor := zlib.NewWriter(&buf)
   425  			if _, err = compressor.Write(body.Bytes()); err != nil {
   426  				return nil, err
   427  			}
   428  			if err = compressor.Close(); err != nil {
   429  				return nil, err
   430  			}
   431  			body = &buf
   432  		} else if headerParams["Content-Encoding"] == "zstd1" {
   433  			body, err = compressZstd(body.Bytes())
   434  			if err != nil {
   435  				return nil, err
   436  			}
   437  		}
   438  		headerParams["Content-Length"] = fmt.Sprintf("%d", body.Len())
   439  		localVarRequest, err = http.NewRequest(method, url.String(), body)
   440  	} else {
   441  		localVarRequest, err = http.NewRequest(method, url.String(), nil)
   442  	}
   443  	if err != nil {
   444  		return nil, err
   445  	}
   446  
   447  	// add header parameters, if any
   448  	if len(headerParams) > 0 {
   449  		headers := http.Header{}
   450  		for h, v := range headerParams {
   451  			headers.Set(h, v)
   452  		}
   453  		localVarRequest.Header = headers
   454  	}
   455  
   456  	// Add the user agent to the request.
   457  	localVarRequest.Header.Add("User-Agent", c.Cfg.UserAgent)
   458  
   459  	if ctx != nil {
   460  		// add context to the request
   461  		localVarRequest = localVarRequest.WithContext(ctx)
   462  
   463  		// Walk through any authentication.
   464  
   465  		// OAuth2 authentication
   466  		if tok, ok := ctx.Value(ContextOAuth2).(oauth2.TokenSource); ok {
   467  			// We were able to grab an oauth2 token from the context
   468  			var latestToken *oauth2.Token
   469  			if latestToken, err = tok.Token(); err != nil {
   470  				return nil, err
   471  			}
   472  
   473  			latestToken.SetAuthHeader(localVarRequest)
   474  		}
   475  
   476  		// Basic HTTP Authentication
   477  		if auth, ok := ctx.Value(ContextBasicAuth).(BasicAuth); ok {
   478  			localVarRequest.SetBasicAuth(auth.UserName, auth.Password)
   479  		}
   480  
   481  		// AccessToken Authentication
   482  		if auth, ok := ctx.Value(ContextAccessToken).(string); ok {
   483  			localVarRequest.Header.Add("Authorization", "Bearer "+auth)
   484  		}
   485  	}
   486  
   487  	if !c.Cfg.Compress {
   488  		// gzip is on by default, so disable it by setting encoding to identity
   489  		localVarRequest.Header.Add("Accept-Encoding", "identity")
   490  	}
   491  	return localVarRequest, nil
   492  }

   494  // Decode unmarshal bytes into an interface
   495  func (c *APIClient) Decode(v interface{}, b []byte, contentType string) (err error) {
   496  	if len(b) == 0 {
   497  		return nil
   498  	}
   499  	if s, ok := v.(*string); ok {
   500  		*s = string(b)
   501  		return nil
   502  	}
   503  	if xmlCheck.MatchString(contentType) {
   504  		if err = xml.Unmarshal(b, v); err != nil {
   505  			return err
   506  		}
   507  		return nil
   508  	}
   509  	if actualObj, ok := v.(interface{ GetActualInstance() interface{} }); ok { // oneOf, anyOf schemas
   510  		if unmarshalObj, ok := actualObj.(interface{ UnmarshalJSON([]byte) error }); ok { // make sure it has UnmarshalJSON defined
   511  			if err = unmarshalObj.UnmarshalJSON(b); err != nil {
   512  				return err
   513  			}
   514  		} else {
   515  			return errors.New("unknown type with GetActualInstance but no unmarshalObj.UnmarshalJSON defined")
   516  		}
   517  	} else if err = Unmarshal(b, v); err != nil { // simple model
   518  		return err
   519  	}
   520  	return nil
   521  }

   543  		delegatedTokenCreds.OrgUUID = creds.OrgUUID

   572  func setBody(body interface{}, contentType string) (bodyBuf *bytes.Buffer, err error) {
   573  	if reflect.ValueOf(body).IsNil() {
   574  		return nil, nil
   575  	}
   576  
   577  	if bodyBuf == nil {
   578  		bodyBuf = &bytes.Buffer{}
   579  	}
   580  
   581  	if reader, ok := body.(io.Reader); ok {
   582  		_, err = bodyBuf.ReadFrom(reader)
   583  	} else if fp, ok := body.(**os.File); ok {
   584  		_, err = bodyBuf.ReadFrom(*fp)
   585  	} else if b, ok := body.([]byte); ok {
   586  		_, err = bodyBuf.Write(b)
   587  	} else if s, ok := body.(string); ok {
   588  		_, err = bodyBuf.WriteString(s)
   589  	} else if s, ok := body.(*string); ok {
   590  		_, err = bodyBuf.WriteString(*s)
   591  	} else if jsonCheck.MatchString(contentType) {
   592  		err = NewEncoder(bodyBuf).Encode(body)
   593  	} else if xmlCheck.MatchString(contentType) {
   594  		err = xml.NewEncoder(bodyBuf).Encode(body)
   595  	}
   596  
   597  	if err != nil {
   598  		return nil, err
   599  	}
   600  
   601  	if bodyBuf.Len() == 0 {
   602  		return nil, fmt.Errorf("invalid body type %s", contentType)
   603  	}
   604  	return bodyBuf, nil
   605  }

   608  func detectContentType(body interface{}) string {
   609  	contentType := "text/plain; charset=utf-8"
   610  	kind := reflect.TypeOf(body).Kind()
   611  
   612  	switch kind {
   613  	case reflect.Struct, reflect.Map, reflect.Ptr:
   614  		contentType = "application/json; charset=utf-8"
   615  	case reflect.String:
   616  		contentType = "text/plain; charset=utf-8"
   617  	default:
   618  		if b, ok := body.([]byte); ok {
   619  			contentType = http.DetectContentType(b)
   620  		} else if kind == reflect.Slice {
   621  			contentType = "application/json; charset=utf-8"
   622  		}
   623  	}
   624  
   625  	return contentType
