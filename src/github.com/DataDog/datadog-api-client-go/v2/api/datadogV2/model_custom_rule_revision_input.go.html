    13  // CustomRuleRevisionInput A revision of a custom static analysis rule as embedded in a rule supplied by a create
    14  // or update request. Nested revisions are sent flat, without a `data`/`type`/`attributes`
    15  // envelope. `id`, `version_id`, `checksum`, `created_at` and `created_by` are server-assigned
    16  // and read-only; they are declared so that a ruleset previously read back can be supplied
    17  // unchanged.
    18  type CustomRuleRevisionInput struct {
    19  	// Rule arguments
    20  	Arguments []Argument `json:"arguments,omitempty"`
    21  	// Rule category
    22  	Category *CustomRuleRevisionAttributesCategory `json:"category,omitempty"`
    23  	// Code checksum
    24  	Checksum *string `json:"checksum,omitempty"`
    25  	// Rule code
    26  	Code *string `json:"code,omitempty"`
    27  	// Creation timestamp
    28  	CreatedAt *time.Time `json:"created_at,omitempty"`
    29  	// Creator identifier
    30  	CreatedBy *string `json:"created_by,omitempty"`
    31  	// Revision creation message
    32  	CreationMessage *string `json:"creation_message,omitempty"`
    33  	// Associated CVE
    34  	Cve datadog.NullableString `json:"cve,omitempty"`
    35  	// Associated CWE
    36  	Cwe datadog.NullableString `json:"cwe,omitempty"`
    37  	// Base64-encoded full description
    38  	Description *string `json:"description,omitempty"`
    39  	// Documentation URL
    40  	DocumentationUrl datadog.NullableString `json:"documentation_url,omitempty"`
    41  	// Revision identifier
    42  	Id *string `json:"id,omitempty"`
    43  	// Whether the revision should be published
    44  	IsPublished *bool `json:"is_published,omitempty"`
    45  	// Whether this is a testing revision
    46  	IsTesting *bool `json:"is_testing,omitempty"`
    47  	// Programming language
    48  	Language *Language `json:"language,omitempty"`
    49  	// Rule severity
    50  	Severity *CustomRuleRevisionAttributesSeverity `json:"severity,omitempty"`
    51  	// Base64-encoded short description
    52  	ShortDescription *string `json:"short_description,omitempty"`
    53  	// Whether to use AI for fixes
    54  	ShouldUseAiFix *bool `json:"should_use_ai_fix,omitempty"`
    55  	// Rule tags
    56  	Tags datadog.NullableList[string] `json:"tags,omitempty"`
    57  	// Rule tests
    58  	Tests []CustomRuleRevisionTest `json:"tests,omitempty"`
    59  	// Tree-sitter query
    60  	TreeSitterQuery *string `json:"tree_sitter_query,omitempty"`
    61  	// Monotonically increasing version number of the revision.
    62  	VersionId *int64 `json:"version_id,omitempty"`
    63  	// UnparsedObject contains the raw value of the object if there was an error when deserializing into the struct
    64  	UnparsedObject map[string]interface{} `json:"-"`
    65  }
