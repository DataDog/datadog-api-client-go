    14  // AnalysisRequestRule A static analysis rule to apply during code analysis. Clients forward complete rule
    15  // objects returned by the rulesets endpoints, so every member of that resource is
    16  // declared here; only `id`, `category`, `checksum`, `language`, `severity`,
    17  // `tree_sitter_query`, `entity_checked`, `regex`, `type` and `code` are read by this
    18  // operation and the rest are ignored. The schema stays open so that any member beyond
    19  // the forwarded rule resource is reported as a promotion candidate rather than
    20  // rejected; it can be closed once that telemetry confirms none remain.
    21  type AnalysisRequestRule struct {
    22  	// The configurable arguments accepted by the rule. Forwarded from the rulesets endpoints; ignored by this operation.
    23  	Arguments []AnalysisRequestRuleArgument `json:"arguments,omitempty"`
    24  	// The category of the rule (for example, `BEST_PRACTICES`, `SECURITY`).
    25  	Category string `json:"category"`
    26  	// A checksum of the rule definition.
    27  	Checksum string `json:"checksum"`
    28  	// The base64-encoded rule implementation code.
    29  	Code string `json:"code"`
    30  	// The date and time when the rule was created. Server-assigned by the rulesets endpoints; ignored by this operation.
    31  	CreatedAt *time.Time `json:"created_at,omitempty"`
    32  	// The identifier of the user or system that created the rule. Server-assigned by the rulesets endpoints; ignored by this operation.
    33  	CreatedBy *string `json:"created_by,omitempty"`
    34  	// The CVE identifier associated with the rule. Forwarded from the rulesets endpoints; ignored by this operation.
    35  	Cve *string `json:"cve,omitempty"`
    36  	// The CWE identifier associated with the rule. Forwarded from the rulesets endpoints; ignored by this operation.
    37  	Cwe *string `json:"cwe,omitempty"`
    38  	// A detailed explanation of what the rule detects. Forwarded from the rulesets endpoints; ignored by this operation.
    39  	Description *string `json:"description,omitempty"`
    40  	// A URL pointing to the rule documentation. Forwarded from the rulesets endpoints; ignored by this operation.
    41  	DocumentationUrl *string `json:"documentation_url,omitempty"`
    42  	// The code entity type checked by the rule, applicable when rule type is `AST_CHECK`.
    43  	EntityChecked datadog.NullableString `json:"entity_checked,omitempty"`
    44  	// The unique identifier of the rule.
    45  	Id string `json:"id"`
    46  	// Whether the rule is published. Forwarded from the rulesets endpoints; ignored by this operation.
    47  	IsPublished *bool `json:"is_published,omitempty"`
    48  	// Whether the rule is in testing mode. Forwarded from the rulesets endpoints; ignored by this operation.
    49  	IsTesting *bool `json:"is_testing,omitempty"`
    50  	// The programming language this rule targets.
    51  	Language string `json:"language"`
    52  	// The date and time when the rule was last modified. Server-assigned by the rulesets endpoints; ignored by this operation.
    53  	LastUpdatedAt *time.Time `json:"last_updated_at,omitempty"`
    54  	// The identifier of the user or system that last updated the rule. Server-assigned by the rulesets endpoints; ignored by this operation.
    55  	LastUpdatedBy *string `json:"last_updated_by,omitempty"`
    56  	// The name of the rule. Forwarded from the rulesets endpoints; ignored by this operation.
    57  	Name *string `json:"name,omitempty"`
    58  	// A base64-encoded regex pattern used by the rule, applicable when rule type is `REGEX`.
    59  	Regex datadog.NullableString `json:"regex,omitempty"`
    60  	// The severity of findings from this rule (for example, `ERROR`, `WARNING`).
    61  	Severity string `json:"severity"`
    62  	// A brief summary of what the rule detects. Forwarded from the rulesets endpoints; ignored by this operation.
    63  	ShortDescription *string `json:"short_description,omitempty"`
    64  	// Whether an AI-generated fix should be offered. Forwarded from the rulesets endpoints; ignored by this operation.
    65  	ShouldUseAiFix *bool `json:"should_use_ai_fix,omitempty"`
    66  	// The test cases associated with the rule. Forwarded from the rulesets endpoints; ignored by this operation.
    67  	Tests []AnalysisRequestRuleTest `json:"tests,omitempty"`
    68  	// The base64-encoded tree-sitter query used by the rule.
    69  	TreeSitterQuery string `json:"tree_sitter_query"`
    70  	// The rule type indicating the detection mechanism (for example, `TREE_SITTER_QUERY`).
    71  	Type string `json:"type"`
    72  	// UnparsedObject contains the raw value of the object if there was an error when deserializing into the struct
    73  	UnparsedObject       map[string]interface{} `json:"-"`
    74  	AdditionalProperties map[string]interface{} `json:"-"`
    75  }
