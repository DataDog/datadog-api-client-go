    14  // GovernanceControlAttributes The attributes of a governance control.
    15  type GovernanceControlAttributes struct {
    16  	// The number of active detections for the control.
    17  	ActiveDetectionsCount int64 `json:"active_detections_count"`
    18  	// The value driver the control is grouped under, such as `security` or `cost`.
    19  	Category string `json:"category"`
    20  	// The time the control configuration was created.
    21  	CreatedAt time.Time `json:"created_at"`
    22  	// The UUID of the user who created the control configuration.
    23  	CreatedBy string `json:"created_by"`
    24  	// A human-readable description of what the control detects.
    25  	Description string `json:"description"`
    26  	// A free-form map of parameter names to their configured values.
    27  	DetectionParameters map[string]interface{} `json:"detection_parameters"`
    28  	// The insight slugs associated with the control.
    29  	Insights []string `json:"insights"`
    30  	// The time of the most recent detection for the control. `null` when there are no detections.
    31  	LastDetectionAt datadog.NullableTime `json:"last_detection_at"`
    32  	// The number of mitigated detections for the control.
    33  	MitigatedDetectionsCount int64 `json:"mitigated_detections_count"`
    34  	// A free-form map of parameter names to their configured values.
    35  	MitigationParameters map[string]interface{} `json:"mitigation_parameters"`
    36  	// The configured mitigation type for the control. Empty when not configured.
    37  	MitigationType string `json:"mitigation_type"`
    38  	// The mitigations available for a control.
    39  	Mitigations []GovernanceControlMitigationDefinition `json:"mitigations"`
    40  	// Human-readable name of the control.

    42  	// The priority of the control, such as `High`.
    43  	Priority string `json:"priority"`
